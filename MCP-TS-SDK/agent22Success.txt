--- SQL query from LLM:
 SELECT 
  STRFTIME(tpep_pickup_datetime, '%Y') AS year,
  STRFTIME(tpep_pickup_datetime, '%m') AS month_num,
  CASE 
    WHEN STRFTIME(tpep_pickup_datetime, '%m') = '01' THEN 'January'
    WHEN STRFTIME(tpep_pickup_datetime, '%m') = '02' THEN 'February'
    WHEN STRFTIME(tpep_pickup_datetime, '%m') = '03' THEN 'March'
    WHEN STRFTIME(tpep_pickup_datetime, '%m') = '04' THEN 'April'
    WHEN STRFTIME(tpep_pickup_datetime, '%m') = '05' THEN 'May'
    WHEN STRFTIME(tpep_pickup_datetime, '%m') = '06' THEN 'June'
    WHEN STRFTIME(tpep_pickup_datetime, '%m') = '07' THEN 'July'
    WHEN STRFTIME(tpep_pickup_datetime, '%m') = '08' THEN 'August'
    WHEN STRFTIME(tpep_pickup_datetime, '%m') = '09' THEN 'September'
    WHEN STRFTIME(tpep_pickup_datetime, '%m') = '10' THEN 'October'
    WHEN STRFTIME(tpep_pickup_datetime, '%m') = '11' THEN 'November'
    WHEN STRFTIME(tpep_pickup_datetime, '%m') = '12' THEN 'December'
  END AS month_name,
  STRFTIME(tpep_pickup_datetime, '%m.%d.%Y') AS formatted_date,
  trip_distance,
  passenger_count,
  total_amount
FROM tripdata
WHERE (STRFTIME(tpep_pickup_datetime, '%Y-%m'), trip_distance) IN (
  SELECT 
    STRFTIME(tpep_pickup_datetime, '%Y-%m'),
    MAX(trip_distance)
  FROM tripdata
  GROUP BY STRFTIME(tpep_pickup_datetime, '%Y-%m')
)
ORDER BY year, month_num
